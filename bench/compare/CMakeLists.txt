# Comparison programs (docs/COMPARACAO.md), one per library, the same scenes in each. Built only with IGUI_BUILD_COMPARE=ON.
# This directory is OUTSIDE the no-std rule (cmake/CheckNoStd.cmake and tools/check.sh scan src/, include/ and adapters/ only):
# the programs of the other libraries use what those libraries need. cmp_igui.cpp itself uses no standard library.
#
#   -DIGUI_COMPARE_IMGUI_DIR=<Dear ImGui source tree>      cmp_imgui (needs SDL2 and OpenGL)
#   -DIGUI_COMPARE_FLTK_CONFIG=<path of fltk-config>        cmp_fltk
#   GTK 3 is found with pkg-config (gtk+-3.0)               cmp_gtk
# Whatever is not found is left out with a message. Always built: libcmp_mcount.so (the LD_PRELOAD heap counter).
enable_language(C)
set(IGUI_COMPARE_IMGUI_DIR "" CACHE PATH "Dear ImGui source tree for the comparison program cmp_imgui (imgui.h, backends/)")
set(IGUI_COMPARE_FLTK_CONFIG "" CACHE FILEPATH "fltk-config of the FLTK to compare with (cmp_fltk)")

find_library(CMP_X11 X11)
find_library(CMP_XTST Xtst)
find_path(CMP_XTEST_INC X11/extensions/XTest.h)
if(NOT CMP_X11 OR NOT CMP_XTST OR NOT CMP_XTEST_INC)
    message(FATAL_ERROR "bench/compare needs libX11 and libXtst (the programs inject real input with XTEST)")
endif()

# plain C, -O2, no sanitizer: it is an instrument
add_library(cmp_mcount SHARED malloc_count.c)
set_target_properties(cmp_mcount PROPERTIES OUTPUT_NAME cmp_mcount)

function(cmp_program name)
    add_executable(${name} ${ARGN} cmp_inject.c)
    target_include_directories(${name} PRIVATE ${CMP_XTEST_INC})
    target_link_libraries(${name} PRIVATE ${CMP_X11} ${CMP_XTST} ${CMAKE_DL_LIBS})
    target_compile_features(${name} PRIVATE cxx_std_17)
endfunction()

cmp_program(cmp_igui cmp_igui.cpp)
target_link_libraries(cmp_igui PRIVATE igui_zen)
# the smallest application of each library, for the size of the binary (strip them; tools/bench_compare.sh does)
add_executable(min_igui min_igui.cpp)
target_link_libraries(min_igui PRIVATE igui_zen)

if(IGUI_COMPARE_IMGUI_DIR AND EXISTS "${IGUI_COMPARE_IMGUI_DIR}/imgui.h")
    find_package(SDL2 CONFIG QUIET)
    find_package(OpenGL QUIET)
    if(NOT SDL2_FOUND)
        find_package(PkgConfig QUIET)
        if(PkgConfig_FOUND)
            pkg_check_modules(CMP_SDL2 QUIET sdl2)
        endif()
    endif()
    if((SDL2_FOUND OR CMP_SDL2_FOUND) AND OpenGL_FOUND)
        set(_im ${IGUI_COMPARE_IMGUI_DIR})
        cmp_program(cmp_imgui cmp_imgui.cpp ${_im}/imgui.cpp ${_im}/imgui_draw.cpp ${_im}/imgui_tables.cpp ${_im}/imgui_widgets.cpp
            ${_im}/backends/imgui_impl_sdl2.cpp ${_im}/backends/imgui_impl_opengl3.cpp)
        target_include_directories(cmp_imgui PRIVATE ${_im} ${_im}/backends)
        add_executable(min_imgui min_imgui.cpp ${_im}/imgui.cpp ${_im}/imgui_draw.cpp ${_im}/imgui_tables.cpp ${_im}/imgui_widgets.cpp
            ${_im}/backends/imgui_impl_sdl2.cpp ${_im}/backends/imgui_impl_opengl3.cpp)
        target_include_directories(min_imgui PRIVATE ${_im} ${_im}/backends)
        foreach(_t cmp_imgui min_imgui)
            if(SDL2_FOUND)
                target_link_libraries(${_t} PRIVATE SDL2::SDL2 OpenGL::GL)
            else()
                target_include_directories(${_t} PRIVATE ${CMP_SDL2_INCLUDE_DIRS})
                target_link_libraries(${_t} PRIVATE ${CMP_SDL2_LIBRARIES} OpenGL::GL)
            endif()
        endforeach()
    else()
        message(STATUS "bench/compare: SDL2 or OpenGL not found, cmp_imgui is not built")
    endif()
else()
    message(STATUS "bench/compare: IGUI_COMPARE_IMGUI_DIR not set, cmp_imgui is not built")
endif()

if(IGUI_COMPARE_FLTK_CONFIG AND EXISTS "${IGUI_COMPARE_FLTK_CONFIG}")
    execute_process(COMMAND ${IGUI_COMPARE_FLTK_CONFIG} --use-images --cxxflags OUTPUT_VARIABLE _fl_cxx OUTPUT_STRIP_TRAILING_WHITESPACE)
    execute_process(COMMAND ${IGUI_COMPARE_FLTK_CONFIG} --use-images --ldflags OUTPUT_VARIABLE _fl_ld OUTPUT_STRIP_TRAILING_WHITESPACE)
    separate_arguments(_fl_cxx UNIX_COMMAND "${_fl_cxx}")
    separate_arguments(_fl_ld UNIX_COMMAND "${_fl_ld}")
    cmp_program(cmp_fltk cmp_fltk.cpp)
    add_executable(min_fltk min_fltk.cpp)
    foreach(_t cmp_fltk min_fltk)
        target_compile_options(${_t} PRIVATE ${_fl_cxx})
        target_link_libraries(${_t} PRIVATE ${_fl_ld})
    endforeach()
else()
    message(STATUS "bench/compare: IGUI_COMPARE_FLTK_CONFIG not set, cmp_fltk is not built")
endif()

find_package(PkgConfig QUIET)
if(PkgConfig_FOUND)
    pkg_check_modules(CMP_GTK3 QUIET gtk+-3.0)
endif()
if(CMP_GTK3_FOUND)
    cmp_program(cmp_gtk cmp_gtk.cpp)
    target_include_directories(cmp_gtk PRIVATE ${CMP_GTK3_INCLUDE_DIRS})
    target_link_directories(cmp_gtk PRIVATE ${CMP_GTK3_LIBRARY_DIRS})
    target_link_libraries(cmp_gtk PRIVATE ${CMP_GTK3_LIBRARIES})
    target_compile_options(cmp_gtk PRIVATE ${CMP_GTK3_CFLAGS_OTHER})
    add_executable(min_gtk min_gtk.cpp)
    target_include_directories(min_gtk PRIVATE ${CMP_GTK3_INCLUDE_DIRS})
    target_link_directories(min_gtk PRIVATE ${CMP_GTK3_LIBRARY_DIRS})
    target_link_libraries(min_gtk PRIVATE ${CMP_GTK3_LIBRARIES})
    target_compile_options(min_gtk PRIVATE ${CMP_GTK3_CFLAGS_OTHER})
else()
    message(STATUS "bench/compare: gtk+-3.0 not found, cmp_gtk is not built")
endif()
